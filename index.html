const displayNotesBtn = document.getElementById('display-notes'); const displayDegreesBtn = document.getElementById('display-degrees'); let showDegrees = false; const degreeMap = { 'C': '1st', 'D': '2nd', 'E': '3rd', 'F': '4th', 'G': '5th', 'A': '6th', 'B': '7th' }; // Function to toggle between displaying Notes or Scale Degrees function toggleDisplay(mode) { showDegrees = mode === 'degrees'; // Set flag based on the selected mode // Update the note button labels noteButtons.forEach(btn => { const note = btn.getAttribute('data-note'); btn.textContent = showDegrees ? degreeMap[note] : note; }); // Toggle button styles displayNotesBtn.classList.toggle('selected', !showDegrees); displayDegreesBtn.classList.toggle('selected', showDegrees); // Update the UI labels scaleLabel.textContent = showDegrees ? 'Diatonic - Major Scale' : 'Diatonic - C Major Scale'; octaveLabel.textContent = showDegrees ? 'Three Octaves' : 'Three Octaves (C4–C5)'; playRefBtn.textContent = showDegrees ? 'Play Reference (Tonic)' : 'Play Reference (C - Tonic)'; promptText.textContent = 'Which note was played?'; } // Add event listeners to toggle between Notes and Degrees displayNotesBtn.addEventListener('click', () => toggleDisplay('notes')); displayDegreesBtn.addEventListener('click', () => toggleDisplay('degrees'));
